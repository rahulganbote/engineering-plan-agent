# PoC Plan Template
# Source: EM Copilot Knowledge Base | Type: plan_template | Version: 3.0
# Synced to Agent 5 (PoC Planner) contract v2.0
#
# CONTRACT FIELDS THIS TEMPLATE DEMONSTRATES:
#   agent_name, citations[], confidence_score, assumptions[], flagged_ambiguities[],
#   poc_hypothesis (specific + falsifiable, not 'validate the approach'),
#   scope_in[] (min 1), scope_out[] (min 2, prevents scope creep),
#   duration_weeks (max 4 - anything longer is an MVP not a PoC),
#   success_criteria[{metric, target_value, measurement_method}] (min 2),
#   team_size (max 4), risk_if_poc_fails, go_nogo_criteria{go, no_go, pivot}

## Output Contract Header
agent_name:          poc_planner
citations:           [PoC_Plan_Template_chunk_N, BRD_chunk_N]
confidence_score:    [0.0-1.0]
assumptions:         [List every assumption made]
flagged_ambiguities: [List ambiguous BRD elements that affect PoC scope]

## PoC Hypothesis
# REQUIRED - Must be specific and falsifiable
# Banned phrases: 'validate the approach', 'test the concept', 'see if it works'
# Must: name the technology, name the measurable outcome, name the BRD risk it addresses
poc_hypothesis: |
  "We believe that [specific technology / integration] can achieve [measurable outcome],
  validating [named BRD risk from risks section]. If true, this de-risks the most
  expensive unknown before full build begins."
  Example: "We believe that LangChain + GPT-4o + Pinecone can generate cross-correlation
  insights rated >= 80% accurate by Domain Expert, validating RISK-01 from BRD
  (LLM narrative quality sufficient for executive decision-making)."

## Scope In
# What WILL be built - minimal viable test only
scope_in:
  - [Single predefined test question or use case - not open-ended]
  - [Minimal data pipeline - hardcoded sample data acceptable]
  - [Core integration under test only]
  - [Simplest output format sufficient to evaluate hypothesis]

## Scope Out
# min 2 items required - explicitly prevents scope creep
scope_out:
  - Full ETL pipeline - use hardcoded sample data
  - User interface or dashboards
  - Authentication or access control layer
  - Production infrastructure or deployment automation
  - Error handling beyond basic logging
  - [Any feature that doesn't directly test the hypothesis]

## Duration
duration_weeks: [1-4]   # MAXIMUM 4 weeks - contract enforces this

## Success Criteria
# min 2 criteria required - each must have: metric, target_value, measurement_method
success_criteria:
  - metric: [What is being measured]
    target_value: [Specific numeric or boolean target - e.g. '>= 80%', '< 15 minutes']
    measurement_method: [How to measure - e.g. 'Automated test over 10 consecutive runs']

  - metric: [Second criterion]
    target_value: [Specific target]
    measurement_method: [Specific measurement method]

  - metric: [Optional third criterion - latency, accuracy, or stakeholder approval]
    target_value: [Target]
    measurement_method: [Method]

## Team
team_size: [1-4]   # More than 4 suggests PoC scope is too large

## Risk If PoC Fails
# REQUIRED - forces consequence thinking, produces better-scoped PoCs
risk_if_poc_fails: |
  If [hypothesis] is disproved, [specific consequence for the project].
  Fallback: [alternative approach if PoC fails].
  Business impact: [what the organization loses or must do differently].

## Go / No-Go Decision Criteria
# All three cases must be specified before PoC begins
go_nogo_criteria:
  go:    "[All criteria that constitute success - specific, not 'good enough']"
  no_go: "[Any criterion whose failure means the approach is not viable]"
  pivot: "[Partial success scenario - what changes to scope or approach to try next]"
